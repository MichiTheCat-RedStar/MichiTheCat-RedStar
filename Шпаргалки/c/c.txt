	--- ОСНОВНЫЕ КОМАНДЫ C ---

 = Компиляция (кратко):

- Скомпилировать программу:

gcc source.c -o program


- Скомпилировать с предупреждениями:

gcc -Wall -Wextra source.c -o program


- Скомпилировать с отладкой (для GDB):

gcc -g source.c -o program


 = Базовый синтаксис:

- Точка входа (функция main):

int main(void) {
    return 0;
}


- Директивы препроцессора (подключение заголовков):

#include <stdio.h>				# стандартные заголовки
#include "myheader.h"			# пользовательские


 = Типы данных:

- Целые: char, short, int, long, long long (со знаком/без)
- Вещественные: float, double, long double
- Указатели: type *ptr
- Структуры: struct name { ... };
- Объединения: union name { ... };
- Перечисления: enum name { ... };
- void: для функций без возврата, или универсальный указатель


 = Управляющие конструкции:

- if / else:

if (condition) { ... } else { ... }


- switch:

switch (expr) { case val: ... break; default: ... }


- for:

for (init; condition; increment) { ... }


- while:

while (condition) { ... }


- do-while:

do { ... } while (condition);


 = Функции:

- Определение:

return_type func_name(params) { ... }


- Прототип (объявление):

return_type func_name(params);


 = Указатели и массивы:

- Объявление указателя:

int *ptr;


- Адрес переменной:

&var


- Разыменование:

*ptr


- Массивы:

int arr[10];				# статический массив
arr[0] = 5;					# доступ по индексу



	--- ПОЛНАЯ ШПАРГАЛКА ПО C ---

 = Структура программы:

- #include (подключение заголовочных файлов)
- #define (макросы)
- Глобальные переменные (не рекомендуется)
- Прототипы функций
- Функция main()
- Определения функций


 = Типы данных (подробно):

- Целые (размер зависит от платформы, но стандарт гарантирует минимум):

char				# 1 байт (обычно)
signed char			# знаковый
unsigned char		# беззнаковый
short int			# не менее 2 байт
unsigned short int
int					# не менее 2 байт (обычно 4)
unsigned int
long int			# не менее 4 байт
unsigned long int
long long int		# не менее 8 байт (C99)
unsigned long long int

- Вещественные:

float				# 4 байта
double				# 8 байт
long double			# 10/16 байт (зависит от платформы)

- Логический (C99, нужен #include <stdbool.h>):

bool				# true/false (на самом деле int)


- void: отсутствие типа, используется для указателей (void*) и функций без возврата.


 = Модификаторы:

- const: неизменяемая переменная
- volatile: значение может измениться вне программы
- register: предложение хранить в регистре (устарело)
- static: для локальных переменных — сохраняет значение между вызовами; для глобальных/функций — ограничивает видимость файлом
- extern: объявление переменной/функции, определённой в другом файле


 = Операторы:

- Арифметические: +, -, *, /, % (остаток)
- Сравнения: ==, !=, <, >, <=, >=
- Логические: &&, ||, !
- Побитовые: &, |, ^, ~, <<, >>
- Присваивание: =, +=, -=, *=, /=, %=, &=, |=, ^=, <<=, >>=
- Инкремент/декремент: ++, -- (префиксные и постфиксные)
- Условный (тернарный): expr ? true_val : false_val
- Оператор sizeof: sizeof(type) или sizeof(expr)
- Оператор приведения: (type) expr
- Оператор адреса: &, разыменование: *
- Доступ к членам структуры: . (для экземпляра), -> (для указателя)
- Индексация массивов: []
- Вызов функции: ()
- Последовательное выполнение: , (запятая)


 = Управляющие конструкции (полно):

- if:

if (condition) statement;
if (condition) statement1; else statement2;
if (cond1) st1; else if (cond2) st2; else st3;


- switch:

switch (expression) {
    case constant1:
        statements;
        break;
    case constant2:
        statements;
        break;
    default:
        statements;
}


- Циклы:

for (init; condition; increment) {
    statements;
}

while (condition) {
    statements;
}

do {
    statements;
} while (condition);


- break и continue: break выходит из цикла/switch, continue переходит к следующей итерации.
- goto: метка и goto label; (не рекомендуется, но иногда полезно для выхода из вложенных циклов)


 = Функции:

- Определение:

return_type function_name(parameter_list) {
    // тело
    return value;
}


- Параметры передаются по значению; для изменения переменной передавайте указатель.
- Функция может не возвращать значение (void).
- Прототипы (объявления) обычно помещаются в заголовочные файлы.
- Рекурсия поддерживается.


 = Массивы:

- Объявление:

int arr[10];					# массив из 10 int
int arr[][5] = {{1,2},{3,4}};	# двумерный массив


- Доступ: arr[index] = value;
- Имя массива — это указатель на первый элемент (arr == &arr[0]).
- Передача массива в функцию: void func(int arr[], int size) или void func(int *arr, int size).


 = Строки (символьные массивы):

- Строка — массив char, заканчивающийся нулевым символом '\0'.
- Объявление:

char str[] = "Hello";			# массив из 6 элементов (включая '\0')
char *str2 = "World";			# указатель на строковой литерал (нельзя изменять)


- Функции для строк (из <string.h>):

strlen(s)					# длина (без '\0')
strcpy(dest, src)			# копирование
strncpy(dest, src, n)		# копирование с ограничением
strcat(dest, src)			# конкатенация
strncat(dest, src, n)		# конкатенация с ограничением
strcmp(s1, s2)				# сравнение (0 — равны)
strncmp(s1, s2, n)			# сравнение первых n символов
strchr(s, c)				# поиск символа
strstr(haystack, needle)	# поиск подстроки
strtok(s, delim)			# разбиение на токены


 = Указатели:

- Определение: type *ptr;
- Присвоение адреса: ptr = &var;
- Разыменование: *ptr = 5;
- Арифметика указателей: ptr++ сдвигает на sizeof(type)
- Указатели на указатели: int **ptr;
- Указатели на функции: return_type (*func_ptr)(params);
- NULL — нулевой указатель (макрос из <stddef.h> или <stdio.h>)


 = Динамическая память (из <stdlib.h>):

- Выделение памяти:

void *malloc(size_t size);		# выделить size байт
void *calloc(size_t num, size_t size);	# выделить num элементов по size байт, инициализирует нулями
void *realloc(void *ptr, size_t new_size);	# изменить размер


- Освобождение:

free(ptr);


- Пример:

int *arr = (int*)malloc(10 * sizeof(int));
if (arr == NULL) { /* ошибка */ }
free(arr);


 = Структуры (struct):

- Определение:

struct Point {
    int x;
    int y;
};


- Объявление переменной:

struct Point p1;
struct Point p2 = {1, 2};
struct Point p3 = {.x = 3, .y = 4};	# C99


- Доступ к полям:

p1.x = 5;
p1.y = 10;


- Указатель на структуру:

struct Point *ptr = &p1;
ptr->x = 7;				# эквивалентно (*ptr).x = 7;


- typedef для упрощения:

typedef struct Point Point_t;
Point_t p4;


 = Объединения (union):

- Определение:

union Data {
    int i;
    float f;
    char str[20];
};


- Используется для хранения разных типов в одной области памяти (размер равен максимуму).


 = Перечисления (enum):

- Определение:

enum Color { RED, GREEN, BLUE };
enum Color c = RED;


- Можно задавать значения: enum { A = 1, B = 2, C = 4 };


 = Препроцессор:

- #include — вставка содержимого файла.
- #define — макрос:

#define PI 3.14159
#define SQUARE(x) ((x)*(x))		# макрос с параметрами


- #undef — отмена определения.
- Условная компиляция:

#ifdef MACRO
    // код
#else
    // код
#endif

#ifndef MACRO
    // код
#endif

#if expression
    // код
#endif

- #pragma — специфичные для компилятора директивы.


 = Стандартные заголовочные файлы (часто используемые):

- <stdio.h> — ввод/вывод (printf, scanf, fopen, fclose, fread, fwrite, fgets, fputs, и т.д.)
- <stdlib.h> — преобразования (atoi, atof), управление памятью (malloc, free), генерация случайных чисел (rand, srand), и т.д.
- <string.h> — строковые функции.
- <math.h> — математические функции (sin, cos, sqrt, pow, ...). При линковке нужен флаг -lm.
- <stdbool.h> — тип bool (C99).
- <stdint.h> — целочисленные типы с фиксированной шириной (int32_t, uint64_t и т.д.)
- <stddef.h> — определения size_t, NULL, offsetof.
- <ctype.h> — функции для работы с символами (isdigit, isalpha, toupper, tolower)
- <time.h> — функции времени (time, clock, localtime, strftime)
- <limits.h> — размеры типов (INT_MAX, ...)
- <float.h> — характеристики чисел с плавающей точкой (FLT_MAX, ...)
- <errno.h> — код ошибки errno.
- <assert.h> — макрос assert для отладки.


 = Функции ввода/вывода (stdio.h):

- Вывод в консоль:

printf("format", ...);		# форматированный вывод
puts("string");				# вывод строки с переводом строки
putchar('c');				# вывод одного символа


- Ввод с консоли:

scanf("format", &var);			# форматированный ввод (требует адреса)
gets()							# устарело, не используйте
fgets(buffer, size, stdin);		# безопасное чтение строки
getchar();						# чтение одного символа


- Форматы для printf/scanf:

%d, %i		# int
%u		# unsigned int
%o, %x, %X	# восьмеричный/шестнадцатеричный
%f, %lf		# float, double
%e, %E		# экспоненциальный
%c		# char
%s		# строка (char*)
%p		# указатель
%%		# символ %
%zu		# size_t (C99)


- Работа с файлами:

FILE *f = fopen("file.txt", "r");	# открыть для чтения
fclose(f);							# закрыть

fprintf(f, "format", ...);			# запись в файл
fscanf(f, "format", &var);			# чтение из файла

fread(buffer, size, count, f);		# бинарное чтение
fwrite(buffer, size, count, f);		# бинарная запись

fseek(f, offset, SEEK_SET);			# переместить указатель
ftell(f);							# текущая позиция
rewind(f);							# в начало

feof(f);							# проверка конца файла
ferror(f);							# проверка ошибки


 = Время (time.h):

time_t t = time(NULL);				# текущее время в секундах с эпохи
struct tm *tm = localtime(&t);		# локальное время
printf("%d-%02d-%02d %02d:%02d:%02d",
       tm->tm_year + 1900,
       tm->tm_mon + 1,
       tm->tm_mday,
       tm->tm_hour,
       tm->tm_min,
       tm->tm_sec);


 = Математические функции (math.h):

sin(x), cos(x), tan(x), asin(x), acos(x), atan(x), atan2(y,x)
sinh(x), cosh(x), tanh(x)
exp(x), log(x), log10(x), pow(x,y), sqrt(x)
ceil(x), floor(x), fabs(x), fmod(x,y)
M_PI и др. константы (но не стандартизованы)


 = Случайные числа (stdlib.h):

srand(time(NULL));			# инициализация генератора
rand();						# число от 0 до RAND_MAX
rand() % N;					# число от 0 до N-1


 = Обработка ошибок:

- errno (из errno.h) устанавливается при ошибках (например, при fopen).
- perror("message"); — выводит сообщение и описание ошибки.
- strerror(errno) — возвращает строку с описанием.


 = Аргументы командной строки (main с параметрами):

int main(int argc, char *argv[]) {
    for (int i = 0; i < argc; i++) {
        printf("arg %d: %s\n", i, argv[i]);
    }
    return 0;
}


 = Компиляция и ссылки (сборка):

- Обычная компиляция:

gcc -c source1.c source2.c				# создать объектные файлы
gcc source1.o source2.o -o program		# линковка


- Одним шагом:

gcc source1.c source2.c -o program


- С подключением библиотек (например, математическая -lm):

gcc program.c -o program -lm


- Использование заголовочных файлов и библиотек из нестандартных путей:

gcc -I/path/to/include -L/path/to/lib -lmylib program.c -o program


 = Полезные макросы и константы:

- __FILE__, __LINE__, __DATE__, __TIME__, __STDC__, __cplusplus (для C++)
- __func__ (имя текущей функции, C99)
- NULL — нулевой указатель
- size_t — беззнаковый тип для размеров


 = Советы:

- Всегда инициализируйте переменные перед использованием.
- Проверяйте возвращаемые значения (malloc, fopen, ...) на NULL.
- Освобождайте динамическую память.
- Используйте const для параметров-указателей, если функция не изменяет данные.
- Избегайте магических чисел, используйте #define или enum.
- Для строк используйте функции с ограничением длины (strncpy, snprintf) во избежание переполнения буфера.
- Включайте заголовочные файлы с прототипами функций, а не определяйте функции в заголовках (кроме inline).
- Для отладки используйте -g и gdb.
- Изучайте документацию (man) для стандартных функций.
