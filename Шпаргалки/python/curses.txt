	--- ОСНОВЫ CURSES ---

 = Назначение:

- Управление текстовым терминалом:

# curses — библиотека для создания текстовых интерфейсов (TUI)
# в терминале: позиционирование курсора, вывод текста, цвета,
# обработка клавиатуры и мыши
# Входит в стандартную библиотеку Python (кроме Windows)


 = Инициализация и завершение:

- Ручной способ:

import curses

stdscr = curses.initscr()		# инициализация, получение окна всего экрана
curses.noecho()					# не отображать вводимые символы
curses.cbreak()					# не ждать Enter при вводе
stdscr.keypad(True)				# включить обработку спецклавиш (стрелки и т.д.)

# ... код программы ...

curses.nocbreak()				# вернуть буферизацию ввода
stdscr.keypad(False)			# отключить обработку спецклавиш
curses.echo()					# вернуть отображение ввода
curses.endwin()					# восстановить терминал


- Рекомендуемый способ (wrapper):

from curses import wrapper

def main(stdscr):
	# stdscr — окно всего экрана
	stdscr.clear()
	stdscr.addstr(0, 0, "Hello, curses!")
	stdscr.refresh()
	stdscr.getch()

wrapper(main)

# wrapper() сам делает init, start_color(), noecho(), cbreak(), keypad()
# и гарантирует восстановление терминала даже при исключении



	--- ОКНА И PADS ---

 = Создание окон:

- Новое окно:

height, width = 5, 40
begin_y, begin_x = 7, 20
win = curses.newwin(height, width, begin_y, begin_x)

# Координаты всегда в порядке (y, x), левый верхний угол — (0, 0)


- Размеры экрана:

curses.LINES		# высота экрана (y)
curses.COLS			# ширина экрана (x)

# Допустимые координаты: от (0, 0) до (LINES-1, COLS-1)


 = Pads:

- Создание pad:

pad = curses.newpad(100, 100)		# высота, ширина
pad.addstr(0, 0, "Текст в pad")
pad.refresh(0, 0, 5, 5, 20, 75)

# refresh(pad_top, pad_left, screen_top, screen_left, screen_bottom, screen_right)



	--- ВЫВОД ТЕКСТА ---

 = Методы окна:

- Добавить строку или символ:

stdscr.addstr("Текст")					# в текущей позиции
stdscr.addstr(5, 10, "Текст")			# в позиции (y=5, x=10)
stdscr.addstr(5, 10, "Текст", attr)		# с атрибутом
stdscr.addch("A")						# один символ
stdscr.addch(5, 10, ord('A'))			# символ по коду


- Перемещение курсора:

stdscr.move(5, 10)			# перейти в (y=5, x=10)


 = Атрибуты:

- Доступные атрибуты:

curses.A_BLINK		# мигающий текст
curses.A_BOLD		# жирный
curses.A_DIM		# тусклый
curses.A_REVERSE	# инвертированный
curses.A_STANDOUT	# лучший доступный режим выделения
curses.A_UNDERLINE	# подчёркнутый


- Пример:

stdscr.addstr(0, 0, "Строка состояния", curses.A_REVERSE)
stdscr.refresh()


 = Обновление экрана:

- refresh():

stdscr.refresh()

# Немедленно обновляет физический экран


- noutrefresh() + doupdate():

win1.noutrefresh()		# пометить окно для обновления
win2.noutrefresh()		# пометить второе окно
curses.doupdate()		# обновить экран один раз

# Эффективнее при нескольких окнах, меньше мерцания

 = Инициализация:

- Включить цвет:

curses.start_color()

if curses.has_colors():
	curses.init_pair(1, curses.COLOR_RED, curses.COLOR_BLACK)
	curses.init_pair(2, curses.COLOR_GREEN, curses.COLOR_WHITE)

# wrapper() вызывает start_color() автоматически


 = Цветовые константы:

- Доступные цвета:

curses.COLOR_BLACK

curses.COLOR_MAGENTA

 = Пары цветов:

- Определение пары:

curses.init_pair(1, curses.COLOR_RED, curses.COLOR_WHITE)
# пара 1 = красный текст на белом фоне
# пара 0 жёстко зашита как белый на чёрном, её нельзя изменить


- Использование пары:

stdscr.addstr(0, 0, "RED ALERT!", curses.color_pair(1))
stdscr.refresh()


- Комбинирование:

stdscr.addstr(0, 0, "Текст", curses.color_pair(1) | curses.A_BOLD)



	--- ВВОД С КЛАВИАТУРЫ ---

 = Получение нажатий:

- getch() — возвращает целое число:

while True:
	c = stdscr.getch()
	if c == ord('q'):
		break
	elif c == curses.KEY_UP:
		pass
	elif c == curses.KEY_DOWN:
		pass

# 0–255 — ASCII-код
# >255 — спецклавиши (KEY_UP, KEY_DOWN, KEY_LEFT, KEY_RIGHT,
# KEY_HOME, KEY_END, KEY_PPAGE, KEY_NPAGE и т.д.)


- getkey() — возвращает строку:

key = stdscr.getkey()
if key == 'q':
	break
elif key == 'KEY_UP':
	pass

# Символы — 1-символьные строки, спецклавиши — 'KEY_UP', '^G' и т.д.


 = Неблокирующий ввод:

- nodelay():

stdscr.nodelay(True)		# не ждать нажатия
c = stdscr.getch()			# curses.ERR (-1), если ничего не нажато


- halfdelay():

curses.halfdelay(5)			# таймаут 0.5 сек (5 десятых)
c = stdscr.getch()			# исключение при таймауте


 = Ввод строки:

- getstr():

curses.echo()
s = stdscr.getstr(0, 0, 15)		# строка до 15 символов, Enter завершает


 = Textbox (curses.textpad):

- Редактируемое поле с Emacs-биндингами:

import curses
from curses.textpad import Textbox, rectangle

def main(stdscr):
	stdscr.addstr(0, 0, "Введите: (Ctrl-G — отправить)")
	editwin = curses.newwin(5, 30, 2, 1)
	rectangle(stdscr, 1, 0, 1+5+1, 1+30+1)
	stdscr.refresh()

	box = Textbox(editwin)
	box.edit()					# редактирование до Ctrl-G
	message = box.gather()		# получить результат

wrapper(main)

# Ctrl-A — в начало строки, Ctrl-E — в конец, Ctrl-K — удалить до конца,
# Ctrl-G — завершить редактирование

 = Частые ошибки:

- Забыли endwin():

# Терминал останется в испорченном состоянии
# Используйте wrapper() для автоматического восстановления


- refresh() после addstr():

# Изменения не видны, пока не вызван refresh() или noutrefresh()+doupdate()
