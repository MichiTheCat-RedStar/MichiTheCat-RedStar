	--- ОСНОВЫ LOGGING ---

 = Назначение:

- Логирование событий:

# Модуль logging — стандартный фреймворк для ведения журналов
# Позволяет писать сообщения разных уровней важности
# в файл, консоль, сеть и другие приёмники
# Входит в стандартную библиотеку Python


 = Импорт:

- Стандартный:

import logging


 = Быстрый старт:

- Простейший пример:

import logging

logging.warning("Это предупреждение")
logging.info("Это не покажется — по умолчанию уровень WARNING")

# По умолчанию вывод в stderr, уровень WARNING


- Настройка уровня:

logging.basicConfig(level=logging.DEBUG)
logging.debug("Теперь видно")



	--- УРОВНИ ЛОГИРОВАНИЯ ---

	--- BASICCONFIG ---

 = Настройка по умолчанию:

- Параметры:

logging.basicConfig(
	level=logging.INFO,						# уровень корневого логгера
	filename='app.log',						# файл (по умолчанию stderr)
	filemode='a',							# 'a' — дозапись, 'w' — перезапись
	format='%(asctime)s %(levelname)s %(message)s',
	datefmt='%Y-%m-%d %H:%M:%S',
	style='%',								# '%', '{' или '$'
	handlers=[...],							# список обработчиков
	force=True,								# сбросить прежние настройки
)


- Пример:

logging.basicConfig(
	level=logging.DEBUG,
	format='%(asctime)s [%(levelname)s] %(name)s: %(message)s',
	datefmt='%H:%M:%S',
)

# basicConfig() действует один раз — повторные вызовы без force игнорируются



	--- HANDLERS ---

 = StreamHandler:

- Вывод в поток:

import logging
import sys

handler = logging.StreamHandler(sys.stdout)	# по умолчанию stderr
handler.setLevel(logging.INFO)
handler.setFormatter(logging.Formatter('%(levelname)s: %(message)s'))

logger = logging.getLogger(__name__)
logger.addHandler(handler)


 = FileHandler:

- Запись в файл:

handler = logging.FileHandler('app.log', mode='a', encoding='utf-8')
handler.setLevel(logging.DEBUG)
logger.addHandler(handler)


 = RotatingFileHandler:

- Ротация по размеру:

from logging.handlers import RotatingFileHandler

handler = RotatingFileHandler(
	'app.log',
	maxBytes=10_000_000,		# 10 МБ
	backupCount=5,				# 5 архивных файлов
	encoding='utf-8',
)
logger.addHandler(handler)


 = TimedRotatingFileHandler:

- Ротация по времени:

from logging.handlers import TimedRotatingFileHandler

handler = TimedRotatingFileHandler(
	'app.log',
	when='midnight',			# 'S', 'M', 'H', 'D', 'midnight', 'W0'–'W6'
	interval=1,
	backupCount=7,
	encoding='utf-8',
)
logger.addHandler(handler)

 = Логирование в JSON:

- Кастомный Formatter:

import json
import logging

class JSONFormatter(logging.Formatter):
	def format(self, record: logging.LogRecord) -> str:
		data = {
			'time': self.formatTime(record),
			'level': record.levelname,
			'name': record.name,
			'message': record.getMessage(),
		}
		if record.exc_info:
			data['exc'] = self.formatException(record.exc_info)
		return json.dumps(data, ensure_ascii=False)


- Подключение:

handler = logging.StreamHandler()
handler.setFormatter(JSONFormatter())
logger.addHandler(handler)


 = Цветной вывод:

- Простой вариант:

class ColorFormatter(logging.Formatter):
	COLORS = {
		'DEBUG': '\033[36m',
		'INFO': '\033[32m',
		'WARNING': '\033[33m',
		'ERROR': '\033[31m',
		'CRITICAL': '\033[35m',
	}
	RESET = '\033[0m'

	def format(self, record):
		color = self.COLORS.get(record.levelname, '')
		msg = super().format(record)
		return f"{color}{msg}{self.RESET}"

	--- ПРИМЕЧАНИЯ ---

 = Рекомендации:

- Один логгер на модуль:

logger = logging.getLogger(__name__)


- Не используйте root для библиотек:

# Библиотеки должны добавлять NullHandler
logger.addHandler(logging.NullHandler())


- Не логируйте через basicConfig в библиотеках:

# Пусть настройка — задача приложения


- Не используйте f-strings:

logger.info("Значение: %s", value)		# правильно
logger.info(f"Значение: {value}")		# неправильно


 = Частые ошибки:

- basicConfig не работает повторно:

# Вызывайте один раз или используйте force=True


- propagate=True в дочерних:

# Сообщения дублируются, если обработчики и на родителе, и на ребёнке


- Логирование секретов:

# Не пишите пароли, токены, персональные данные
