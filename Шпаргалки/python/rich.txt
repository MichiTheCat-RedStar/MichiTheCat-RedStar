	--- ОСНОВЫ RICH ---

 = Установка:

- Базовый пакет:

pip install rich


- С поддержкой Jupyter:

pip install "rich[jupyter]"


- Встроенное демо:

python -m rich


 = Назначение:

- Богатый вывод в терминал:

# Rich — библиотека для цветного текста, таблиц, панелей,
# прогресс-баров, подсветки синтаксиса, Markdown и многого другого
# в терминале


 = Быстрый старт:

- Замена print():

from rich import print

# Переопределяет встроенный print
print("[bold red]Hello[/] World!")		# Console Markup
print({"key": "value", "n": 42})		# авто pretty-print


- Использование объекта Console (рекомендуется):

from rich.console import Console

console = Console()
console.print("Hello", style="bold")
console.log("debug info")				# с отметкой времени
console.rule("[bold blue]Section")		# горизонтальная линия



	--- CONSOLE API ---

 = Создание консоли:

- Параметры конструктора:

console = Console()							# автоопределение
console = Console(width=120)				# фиксированная ширина
console = Console(color_system="truecolor")	# 16M цветов
console = Console(stderr=True)				# вывод в stderr
console = Console(record=True)				# включить экспорт
console = Console(force_terminal=True)		# цвет в пайпах
console = Console(style="white on blue")	# стиль по умолчанию


 = Вывод:

- Стилизованный вывод:

console.print("Hello", style="bold red")
console.print("Left", justify="left")
console.print("Center", justify="center")
console.print("Right", justify="right")
console.print("[link=https://example.com]Click[/link]")


- Pretty JSON:

console.print_json('{"a": 1, "b": [2, 3]}')


- Логирование:

console.log("Starting process")			# с отметкой времени
console.log("x =", x, log_locals=True)	# показать локальные переменные


- Горизонтальные линии:

console.rule()								# простая линия
console.rule("[bold red]Chapter 1")			# с заголовком
console.rule(style="cyan", align="left")	# выравнивание


- Ввод:

name = console.input("[bold]Name? [/]")	# стилизованный prompt


- Спиннер (статус):

with console.status("Working..."):
	do_work()

with console.status("Loading", spinner="dots"):
	do_work()


- Пейджер:

with console.pager():					# системный пейджер (less, more)
	console.print(long_text)

with console.pager(styles=True):		# сохранить цвета
	console.print(long_text)


- Альтернативный экран:

with console.screen():
	console.print("Full screen")

	--- ПРОГРЕСС И СТАТУС ---

 = Прогресс-бары:

- Базовый прогресс:

from rich.progress import track
import time

for i in track(range(100), description="Обработка..."):
	time.sleep(0.01)


- Ручное управление:

from rich.progress import Progress

with Progress() as progress:
	task = progress.add_task("Загрузка...", total=100)
	while not progress.finished:
		progress.update(task, advance=1)
		time.sleep(0.01)


- Несколько задач:

with Progress() as progress:
	task1 = progress.add_task("Задача 1", total=100)
	task2 = progress.add_task("Задача 2", total=200)
	while not progress.finished:
		progress.update(task1, advance=1)
		progress.update(task2, advance=2)
		time.sleep(0.01)


- Кастомизация:

from rich.progress import (
	Progress,
	SpinnerColumn,
	TextColumn,
	BarColumn,
	TimeRemainingColumn,
)

with Progress(
	SpinnerColumn(),
	TextColumn("[progress.description]{task.description}"),
	BarColumn(),
	TextColumn("[progress.percentage]{task.percentage:>3.0f}%"),
	TimeRemainingColumn(),
) as progress:
	task = progress.add_task("Работа...", total=100)

	--- ЛОГИРОВАНИЕ ---

 = RichHandler:

- Настройка:

import logging
from rich.logging import RichHandler

FORMAT = "%(message)s"

logging.basicConfig(
	level="NOTSET",
	format=FORMAT,
	datefmt="[%X]",
	handlers=[RichHandler()],
)

log = logging.getLogger("rich")
log.info("Hello, World!")


- С разметкой:

log.error("[bold red blink]Сервер выключается![/]",
	extra={"markup": True})


- Отключить подсветку:

log.error("123 не будет подсвечен",
	extra={"highlighter": None})


 = Обработка исключений:

- Красивые исключения в логах:

logging.basicConfig(
	level="NOTSET",
	format="%(message)s",
	datefmt="[%X]",
	handlers=[RichHandler(rich_tracebacks=True)],
)

log = logging.getLogger("rich")

try:
	1 / 0
except Exception:
	log.exception("Ошибка деления!")


- Подавление фреймов:

from rich.logging import RichHandler
import click

logging.basicConfig(
	level="NOTSET",
	format="%(message)s",
	datefmt="[%X]",
	handlers=[RichHandler(
		rich_tracebacks=True,
		tracebacks_suppress=[click],
	)],
)

	--- LIVE DISPLAY И LAYOUT ---

 = Live:

- Обновление в реальном времени:

from rich.live import Live
from rich.table import Table
import time

def generate_table() -> Table:
	table = Table()
	table.add_column("Время")
	table.add_row(str(time.time()))
	return table

with Live(generate_table(), refresh_per_second=4) as live:
	for _ in range(20):
		time.sleep(0.25)
		live.update(generate_table())


 = Layout:

- Разделение терминала:

from rich.layout import Layout
from rich.console import Console

console = Console()
layout = Layout()

layout.split_column(
	Layout(name="header", size=3),		# фиксированная высота
	Layout(name="body", ratio=1),		# гибкая высота
	Layout(name="footer", size=3),
)

layout["body"].split_row(
	Layout(name="left"),				# левая панель
	Layout(name="right"),				# правая панель
)

layout["header"].update("Заголовок")
layout["left"].update("Левая панель")
layout["right"].update("Правая панель")

console.print(layout)
