 = Асинхронный клиент:

- Базовый асинхронный чат:

import asyncio
from ollama import AsyncClient

async def chat():
    message = {'role': 'user', 'content': 'Почему небо голубое?'}
    response = await AsyncClient().chat(
        model='llama2',
        messages=[message]
    )
    print(response['message']['content'])

asyncio.run(chat())


- Асинхронный стриминг:

import asyncio
from ollama import AsyncClient

async def chat():
    message = {'role': 'user', 'content': 'Напиши стих'}
    async for part in await AsyncClient().chat(
        model='llama2',
        messages=[message],
        stream=True
    ):
        print(part['message']['content'], end='', flush=True)

asyncio.run(chat())

 = Структурированный вывод (JSON Schema):

- С Pydantic:

import json
from pydantic import BaseModel
import ollama

class Task(BaseModel):
    title: str
    priority: int

schema = Task.model_json_schema()

output = ollama.chat(
    model='llama2',
    format=schema,
    options={'temperature': 0},
    messages=[{
        'role': 'user',
        'content': "Извлеки задачу: 'Исправить баг до пятницы'"
    }]
)

task = Task(**json.loads(output['message']['content']))
print(task.title, task.priority)


- С простым JSON-словарём:

schema = {
    "type": "object",
    "properties": {
        "name": {"type": "string"},
        "age": {"type": "integer"}
    },
    "required": ["name", "age"]
}

output = ollama.chat(
    model='llama2',
    format=schema,
    messages=[{'role': 'user', 'content': 'Данные о человеке: Иван, 30 лет'}]
)
print(output['message']['content'])

- Вызов функции из ответа модели:

available_functions = {
    'add_two_numbers': add_two_numbers,
}

for tool in response.message.tool_calls or []:
    function_to_call = available_functions.get(tool.function.name)
    if function_to_call:
        print('Результат:', function_to_call(**tool.function.arguments))
    else:
        print('Функция не найдена:', tool.function.name)


- Передача существующих функций (например, requests):

import ollama
import requests

available_functions = {
    'request': requests.request,
}

response = ollama.chat(
    model='llama3.1',
    messages=[{
        'role': 'user',
        'content': 'Получи страницу ollama.com'
    }],
    tools=[requests.request]
)

for tool in response.message.tool_calls or []:
    function_to_call = available_functions.get(tool.function.name)
    if function_to_call == requests.request:
        resp = function_to_call(
            method=tool.function.arguments.get('method'),
            url=tool.function.arguments.get('url')
        )
        print(resp.text)

- Передача функции как инструмента:

import json
import requests

response = requests.post(
    f'{BASE_URL}/chat',
    json={
        'model': 'llama3.1',
        'messages': [{'role': 'user', 'content': 'Сколько будет 10 + 10?'}],
        'tools': [{
            'type': 'function',
            'function': {
                'name': 'add_two_numbers',
                'description': 'Сложить два числа',
                'parameters': {
                    'type': 'object',
                    'properties': {
                        'a': {'type': 'integer', 'description': 'Первое число'},
                        'b': {'type': 'integer', 'description': 'Второе число'}
                    },
                    'required': ['a', 'b']
                }
            }
        }],
        'stream': False
    }
)

data = response.json()
tool_calls = data.get('message', {}).get('tool_calls', [])

available_functions = {
    'add_two_numbers': add_two_numbers,
}

for tool in tool_calls:
    function_to_call = available_functions.get(tool['function']['name'])
    if function_to_call:
        args = tool['function']['arguments']
        if isinstance(args, str):
            args = json.loads(args)
        print('Результат:', function_to_call(**args))
    else:
        print('Функция не найдена:', tool['function']['name'])


- Передача существующих функций (например, requests):

import requests

available_functions = {
    'request': requests.request,
}

response = requests.post(
    f'{BASE_URL}/chat',
    json={
        'model': 'llama3.1',
        'messages': [{
            'role': 'user',
            'content': 'Получи страницу ollama.com'
        }],
        'tools': [{
            'type': 'function',
            'function': {
                'name': 'request',
                'description': 'Выполнить HTTP-запрос',
                'parameters': {
                    'type': 'object',
                    'properties': {
                        'method': {'type': 'string'},
                        'url': {'type': 'string'}
                    },
                    'required': ['method', 'url']
                }
            }
        }],
        'stream': False
    }
)

data = response.json()
tool_calls = data.get('message', {}).get('tool_calls', [])

for tool in tool_calls:
    if tool['function']['name'] == 'request':
        args = tool['function']['arguments']
        if isinstance(args, str):
            args = json.loads(args)
        resp = requests.request(
            method=args.get('method'),
            url=args.get('url')
        )
        print(resp.text)
