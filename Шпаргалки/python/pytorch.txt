- Минимальный модуль:
class MLP(nn.Module):
    def __init__(self, in_features: int, hidden: int, out_features: int):
        super().__init__()
        self.fc1 = nn.Linear(in_features, hidden)
        self.act = nn.ReLU()
        self.fc2 = nn.Linear(hidden, out_features)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.fc1(x)
        x = self.act(x)
        x = self.fc2(x)
        return x

 = EMA / model averaging:
class EMA:
    def __init__(self, model, decay=0.999):
        self.model = model
        self.decay = decay
        self.shadow = {k: v.clone().detach() for k, v in model.state_dict().items()}

    @torch.no_grad()
    def update(self):
        for k, v in self.model.state_dict().items():
            if v.dtype.is_floating_point:
                self.shadow[k].mul_(self.decay).add_(v.detach(), alpha=1 - self.decay)
            else:
                self.shadow[k] = v.clone()

    def apply_to(self, model):
        model.load_state_dict(self.shadow)

 = Частые ошибки:
- Size mismatch:
print(x.shape, logits.shape, y.shape)
# CrossEntropyLoss: logits (N, C), targets (N,) dtype long
# BCEWithLogitsLoss: logits (N, C), targets float той же формы
# flatten перед Linear: x = x.flatten(1) или x.view(x.size(0), -1)


- CUDA error:
device mismatch				# model, x, y, optimizer state на одном device
OOM							# reduce batch, fp16/bf16, checkpointing, empty_cache
illegal memory access		# проверить индексы, маски, NaN, invalid gather/scatter
non_blocking без pin_memory	# может не дать эффект или гонки


- NaN/Inf loss:
print(loss.item(), torch.isnan(loss).item(), torch.isinf(loss).item())
проверить learning rate
gradient clipping
добавить eps в log/division
проверить targets dtype для CrossEntropyLoss
проверить нормализацию данных
проверить, что loss не делится на zero-size batch


 = Проверка градиентов:
- После backward:
for name, p in model.named_parameters():
    if p.grad is not None:
        print(name, p.grad.abs().mean().item(), p.grad.norm().item())
    else:
        print(name, "no grad")


- Типичные причины no grad:
модуль не part of model				# создан вне __init__
операция вне autograd				# .detach(), .item(), numpy
requires_grad=False					# замороженные параметры
loss не связан с output				# забыл использовать logits в criterion
