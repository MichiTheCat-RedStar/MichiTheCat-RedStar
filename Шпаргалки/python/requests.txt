	--- ОСНОВЫ REQUESTS ---

 = Назначение:

- HTTP-клиент:

# requests — простая и удобная библиотека для HTTP-запросов
# Поддерживает GET, POST, PUT, DELETE, PATCH, HEAD, OPTIONS
# Автоматически обрабатывает JSON, cookies, сессии, редиректы
# Не входит в стандартную библиотеку Python


 = Установка:

- Через pip:

pip install requests


 = Импорт:

- Стандартный:

import requests


 = Быстрый старт:

- GET-запрос:

import requests

response = requests.get('https://api.github.com')
print(response.status_code)
print(response.json())

	--- ОШИБКИ И ИСКЛЮЧЕНИЯ ---

 = Базовое исключение:

- requests.exceptions.RequestException:

# Все исключения requests наследуются от RequestException


 = Частые исключения:

- ConnectionError:

# Ошибка соединения


- HTTPError:

# raise_for_status() вызвал ошибку


- Timeout:

# Таймаут


- TooManyRedirects:

# Слишком много редиректов


- URLRequired:

# URL не указан


- MissingSchema:

# URL без схемы (http://, https://)


- InvalidSchema:

# Неподдерживаемая схема


- InvalidURL:

# Некорректный URL


- InvalidHeader:

# Некорректный заголовок


- ChunkedEncodingError:

# Ошибка chunked-кодирования


- ContentDecodingError:

# Ошибка декодирования контента


- StreamConsumedError:

# Поток уже прочитан


- RetryError:

# Ошибка после повторных попыток


- UnrewindableBodyError:

# Невозможно перемотать тело


 = Обработка:

- try/except:

import requests
from requests.exceptions import HTTPError, Timeout, ConnectionError

try:
	response = requests.get('https://example.com', timeout=5)
	response.raise_for_status()
except HTTPError as e:
	print(f'HTTP error: {e}')
except Timeout:
	print('Таймаут')
except ConnectionError:
	print('Ошибка соединения')
except requests.exceptions.RequestException as e:
	print(f'Другая ошибка: {e}')

	--- АДАПТЕРЫ И RETRY ---

 = HTTPAdapter:

- Создание:

from requests.adapters import HTTPAdapter

adapter = HTTPAdapter(max_retries=3, pool_connections=10, pool_maxsize=10)
session = requests.Session()
session.mount('https://', adapter)
session.mount('http://', adapter)


 = Retry:

- urllib3.util.retry.Retry:

from urllib3.util.retry import Retry

retry = Retry(
	total=3,
	read=3,
	connect=3,
	backoff_factor=0.5,
	status_forcelist=[500, 502, 503, 504],
	method_whitelist=frozenset(['GET', 'POST']),
)
adapter = HTTPAdapter(max_retries=retry)


 = Пример:

- Полный:

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

session = requests.Session()
retry = Retry(total=5, backoff_factor=0.1, status_forcelist=[500, 502, 503, 504])
adapter = HTTPAdapter(max_retries=retry)
session.mount('http://', adapter)
session.mount('https://', adapter)
response = session.get('https://example.com')



	--- ПРИМЕЧАНИЯ ---

 = Совместимость:

- Версии Python:

# requests 2.32: Python 3.8+
# requests 2.31: Python 3.7+
# requests 2.28: Python 3.7+
# requests 2.27: Python 3.6+
# Рекомендуется Python 3.8+


 = Частые ошибки:

- Забыли timeout:

# Без timeout запрос может висеть бесконечно
# Всегда указывайте timeout


- verify=False:

# Отключает проверку SSL, небезопасно
# Используйте только для отладки


- Сессия не закрыта:

# Используйте with requests.Session() as session:


- Content-Type:

# json= устанавливает application/json
# data= устанавливает application/x-www-form-urlencoded
# files= устанавливает multipart/form-data


- Кодировка:

# response.text использует encoding из заголовков
# Если кодировка неверна: response.encoding = 'utf-8'


 = Рекомендации:

- Используйте сессии:

# Для нескольких запросов к одному хосту
# Сохраняет cookies, заголовки, соединения


- Всегда указывайте timeout:

# Защита от зависаний


- Обрабатывайте исключения:

# try/except requests.exceptions.RequestException


- Используйте raise_for_status():

# Для проверки статуса


- Для больших файлов:

# stream=True + iter_content()


- Для повторных попыток:

# HTTPAdapter + Retry


- Для отладки:

# response.request.headers, response.request.body


- Для JSON:

# response.json() вместо json.loads(response.text)


- Для проверки SSL:

# Не отключайте verify без необходимости


- Для прокси:

# Учитывайте переменные окружения
